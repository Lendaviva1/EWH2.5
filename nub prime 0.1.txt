-- LÓGICA DE COMBATE (AIMBOT ESTÁVEL & WALL-LOCK)
-- ==========================================

local function GetBestTarget()
    local Closest = nil
    local MaxDist = Settings.AimbotFOV

    if CurrentLockTarget and CurrentLockTarget.Character and CurrentLockTarget.Character:FindFirstChild("Head") then
        local root = CurrentLockTarget.Character:FindFirstChild("HumanoidRootPart")
        if root then
            local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
            if onScreen then
                local mousePos = UserInputService:GetMouseLocation()
                local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                if dist < MaxDist then return CurrentLockTarget end
            end
        end
    end

    for _, Player in pairs(Players:GetPlayers()) do
        if Player == LocalPlayer then continue end
        if Controls.TeamCheck and Player.Team == LocalPlayer.Team then continue end
        local Root = Player.Character and Player.Character:FindFirstChild("HumanoidRootPart")
        local Hum = Player.Character and Player.Character:FindFirstChild("Humanoid")
        if Root and Hum and Hum.Health > 0 then
            local Pos, OnScreen = Camera:WorldToViewportPoint(Root.Position)
            if OnScreen then
                local MousePos = UserInputService:GetMouseLocation()
                local Dist = (Vector2.new(Pos.X, Pos.Y) - MousePos).Magnitude
                if Dist < MaxDist then
                    MaxDist = Dist
                    Closest = Player
                end
            end
        end
    end
    return Closest
end

RunService.RenderStepped:Connect(function()
    -- 1. Lógica do FOV Inteligente
    local TargetInFOV = GetBestTarget()
    FOVCircle.Visible = Controls.ShowFOV
    FOVCircle.Position = UserInputService:GetMouseLocation()
    
    if UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) and Controls.Aimbot_Enabled and TargetInFOV then
        FOVCircle.Color = Settings.FOV_Locked
        pulseStep = pulseStep + 0.1
        FOVCircle.Radius = Settings.AimbotFOV + math.sin(pulseStep) * 5
    elseif TargetInFOV then
        FOVCircle.Color = Settings.FOV_Target
        FOVCircle.Radius = Settings.AimbotFOV
    else
        FOVCircle.Color = Settings.FOV_Empty
        FOVCircle.Radius = Settings.AimbotFOV
    end

    -- 2. Aimbot Execution (Wall-Lock + Smoothness 1)
    if Controls.Aimbot_Enabled and UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
        local Target = TargetInFOV
        if Target and Target.Character and Target.Character:FindFirstChild("Head") then
            CurrentLockTarget = Target
            local HeadPos = Target.Character.Head.Position
            local CurrentCF = Camera.CFrame
            local TargetCF = CFrame.new(CurrentCF.Position, HeadPos)
            Camera.CFrame = CurrentCF:Lerp(TargetCF, Settings.AimbotSmoothness)
        else
            CurrentLockTarget = nil
        end
    else
        CurrentLockTarget = nil
    end

    -- 3. ESP Híbrido e Target Window
    local ClosestForUI = nil
    local ShortestUI = math.huge

    for _, Player in pairs(Players:GetPlayers()) do
        if Player == LocalPlayer then continue end
        local isEnemy = not (Controls.TeamCheck and Player.Team == LocalPlayer.Team)
        if not isEnemy then 
            if Chams_Cache[Player] then Chams_Cache[Player].Parent = nil end
            if Box_Cache[Player] then Box_Cache[Player].Visible = false end
            continue 
        end

        local Character = Player.Character
        local Root = Character and Character:FindFirstChild("HumanoidRootPart")
        local Hum = Character and Character:FindFirstChild("Humanoid")
        local Head = Character and Character:FindFirstChild("Head")

        if not Root or not Hum or not Head or Hum.Health <= 0 then
            if Chams_Cache[Player] then Chams_Cache[Player].Parent = nil end
            if Box_Cache[Player] then Box_Cache[Player].Visible = false end
            continue
        end

        if Controls.ESP_Enabled then
            local distance = (Camera.CFrame.Position - Root.Position).Magnitude
            local visible = (function()
                local rp = RaycastParams.new()
                rp.FilterDescendantsInstances = {LocalPlayer.Character, Character}
                rp.FilterType = Enum.RaycastFilterType.Blacklist
                return workspace:Raycast(Camera.CFrame.Position, Root.Position - Camera.CFrame.Position, rp) == nil
            end)()
            local color = visible and Settings.VisibleColor or Settings.HiddenColor

            if distance < Settings.HybridDistance then
                if Box_Cache[Player] then Box_Cache[Player].Visible = false end
                local hl = Chams_Cache[Player] or Instance.new("Highlight")
                hl.FillColor = color; hl.FillTransparency = 0.5; hl.OutlineColor = Color3.fromRGB(255,255,255); hl.Adornee = Character; hl.Parent = Character
                Chams_Cache[Player] = hl
            else
                if Chams_Cache[Player] then Chams_Cache[Player].Parent = nil end
                if not Box_Cache[Player] then 
                    local b = Drawing.new("Square")
                    b.Visible = false; b.Thickness = 1; b.Filled = false
                    Box_Cache[Player] = b
                end
                local box = Box_Cache[Player]
                local pos, onScreen = Camera:WorldToViewportPoint(Root.Position)
                if onScreen then
                    local size = (1000 / distance) * 3
                    box.Visible = true; box.Size = Vector2.new(size, size * 1.5); box.Position = Vector2.new(pos.X - size/2, pos.Y - size/2); box.Color = color
                else
                    box.Visible = false
                end
            end

            local pos, onScreen = Camera:WorldToViewportPoint(Root.Position)
            if onScreen then
                local distToMouse = (Vector2.new(pos.X, pos.Y) - UserInputService:GetMouseLocation()).Magnitude
                if distToMouse < ShortestUI then
                    ShortestUI = distToMouse
                    ClosestForUI = Player
                end
            end
        end
    end

    if Controls.TargetInfo_Enabled and ClosestForUI then
        local TChar = ClosestForUI.Character
        local TRoot = TChar and TChar:FindFirstChild("HumanoidRootPart")
        local THum = TChar and TChar:FindFirstChild("Humanoid")
        if TRoot and THum then
            local pos, onScreen = Camera:WorldToViewportPoint(TRoot.Position)
            if onScreen then
                InfoWindow.Visible = true
                InfoWindow.Position = UDim2.new(0, pos.X + 20, 0, pos.Y - 20)
                nameVal.Text = ClosestForUI.Name
                distVal.Text = string.format("%.1f m", (Camera.CFrame.Position - TRoot.Position).Magnitude / 3.57)
                speedVal.Text = math.floor(TRoot.Velocity.Magnitude * 3.6) .. " km/h"
                hpVal.Text = math.floor(THum.Health) .. " HP"
                HPBar.Size = UDim2.new(math.clamp(THum.Health / THum.MaxHealth, 0, 1), 0, 1, 0)
            else InfoWindow.Visible = false end
        else InfoWindow.Visible = false end
    else InfoWindow.Visible = false end
end)

    if Box_Cache[P] then Box_Cache[P]:Remove() end
end)
